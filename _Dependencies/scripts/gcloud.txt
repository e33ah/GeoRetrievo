# create bucket for country data
gcloud storage buckets create gs://country_data_01 \
    --default-storage-class=STANDARD \
    --location=US-CENTRAL1 \
    --uniform-bucket-level-access \
    --public-access-prevention

# upload country data to bucket
gcloud storage cp ./GeoRetrievo.API/Data/countries.json \
  gs://country_data_01/countries.json

# create service account for GeoRetrievo API
gcloud iam service-accounts create georetrievo-api-svc \
  --display-name="GeoRetrievo API svc"

# grant service account access to bucket
gcloud storage buckets add-iam-policy-binding \
  gs://country_data_01 \
  --member="serviceAccount:georetrievo-api-svc@georetrievo.iam.gserviceaccount.com" \
  --role="roles/storage.objectViewer"

# enable required services
gcloud services enable \
  run.googleapis.com \
  cloudbuild.googleapis.com \
  artifactregistry.googleapis.com 

# terrform import commands for required services
terraform import 'google_project_service.required["run.googleapis.com"]' 'georetrievo/run.googleapis.com'
terraform import 'google_project_service.required["cloudbuild.googleapis.com"]' 'georetrievo/cloudbuild.googleapis.com'
terraform import 'google_project_service.required["artifactregistry.googleapis.com"]' 'georetrievo/artifactregistry.googleapis.com'
terraform import 'google_project_service.required["storage.googleapis.com"]' 'georetrievo/storage.googleapis.com'
terraform import 'google_project_service.required["iam.googleapis.com"]' 'georetrievo/iam.googleapis.com'
terraform import google_storage_bucket_iam_member.country_data_reader "country_data_01 roles/storage.objectViewer serviceAccount:georetrievo-api-svc@georetrievo.iam.gserviceaccount.com"
