# create bucket for country data
gcloud storage buckets create gs://country_data_01 \
    --default-storage-class=STANDARD \
    --location=US-CENTRAL1 \
    --uniform-bucket-level-access \
    --public-access-prevention

# upload country data to bucket
gcloud storage cp ./GeoRetrievo.API/Data/countries.json \
  gs://country_data_01/countries.json

# create service account for GeoRetrievo API
gcloud iam service-accounts create georetrievo-api-svc \
  --display-name="GeoRetrievo API svc"

# grant service account access to bucket
gcloud storage buckets add-iam-policy-binding \
  gs://country_data_01 \
  --member="serviceAccount:georetrievo-api-svc@georetrievo.iam.gserviceaccount.com" \
  --role="roles/storage.objectViewer"

# enable required services
gcloud services enable \
  run.googleapis.com \
  cloudbuild.googleapis.com \
  artifactregistry.googleapis.com \
  cloudresourcemanager.googleapis.com

# create bucket for terraform state
gcloud storage buckets create gs://georetrievo-terraform-state --location=US-CENTRAL1 --uniform-bucket-level-access  --public-access-prevention
gcloud storage buckets update gs://georetrievo-terraform-state --versioning

# terrform import commands for required services
terraform import 'google_project_service.required["run.googleapis.com"]' 'georetrievo/run.googleapis.com'
terraform import 'google_project_service.required["cloudbuild.googleapis.com"]' 'georetrievo/cloudbuild.googleapis.com'
terraform import 'google_project_service.required["artifactregistry.googleapis.com"]' 'georetrievo/artifactregistry.googleapis.com'
terraform import 'google_project_service.required["storage.googleapis.com"]' 'georetrievo/storage.googleapis.com'
terraform import 'google_project_service.required["iam.googleapis.com"]' 'georetrievo/iam.googleapis.com'
terraform import google_storage_bucket_iam_member.country_data_reader "country_data_01 roles/storage.objectViewer serviceAccount:georetrievo-api-svc@georetrievo.iam.gserviceaccount.com"

# create service account for github deploy
gcloud iam service-accounts create github-deploy-svc \
  --display-name="GitHub Deploy svc"

# grant service account access to workload identity pool
gcloud iam service-accounts add-iam-policy-binding github-deploy-svc@georetrievo.iam.gserviceaccount.com \
  --role="roles/iam.workloadIdentityUser" \
  --member="principalSet://iam.googleapis.com/projects/194067282433/locations/global/workloadIdentityPools/github-pool-01/attribute.repository/e33ah/GeoRetrievo"

 # grant service account access to deploy to Cloud Run and Artifact Registry
 gcloud projects add-iam-policy-binding georetrievo \
  --member="serviceAccount:github-deploy-svc@georetrievo.iam.gserviceaccount.com" \
  --role="roles/run.admin"

  gcloud projects add-iam-policy-binding georetrievo \
  --member="serviceAccount:github-deploy-svc@georetrievo.iam.gserviceaccount.com" \
  --role="roles/artifactregistry.writer"


$DEPLOY_SA = "github-deploy-svc@georetrievo.iam.gserviceaccount.com"
$RUNTIME_SA = "georetrievo-api-svc@georetrievo.iam.gserviceaccount.com"

# Let Terraform enable the required Google Cloud APIs.
gcloud projects add-iam-policy-binding georetrievo --member="serviceAccount:$DEPLOY_SA" --role="roles/serviceusage.serviceUsageAdmin"

# Let Terraform create/manage the Artifact Registry repository as well as push images.
gcloud projects add-iam-policy-binding georetrievo --member="serviceAccount:$DEPLOY_SA" --role="roles/artifactregistry.admin"

# Give Terraform access to its dedicated remote-state bucket.
gcloud storage buckets add-iam-policy-binding gs://georetrievo-terraform-state --member="serviceAccount:$DEPLOY_SA" --role="roles/storage.admin"

# Create a narrow role for managing the country-data bucket's IAM policy.
gcloud iam roles create georetrievoBucketIamManager --project=georetrievo --title="GeoRetrievo bucket IAM manager" --description="Allows Terraform to read and update bucket IAM policy" --permissions="storage.buckets.get,storage.buckets.getIamPolicy,storage.buckets.setIamPolicy" --stage=GA

# Let Terraform manage the country-data bucket IAM grant without granting access to its objects.
gcloud storage buckets add-iam-policy-binding gs://country_data_01 --member="serviceAccount:$DEPLOY_SA" --role="projects/georetrievo/roles/georetrievoBucketIamManager"

# Let Terraform inspect and deploy Cloud Run using the existing runtime service account.
gcloud iam service-accounts add-iam-policy-binding $RUNTIME_SA --project=georetrievo --member="serviceAccount:$DEPLOY_SA" --role="roles/iam.serviceAccountUser"
